# 数据结构第三次作业

## 3.29

```c
typedef struct{
    Elemtype *base;
    int front,rear;
    int tag;//初始化应置为0
}SqQueue;

Status EnQueue(SqQueue *q,Elemtype e){
    if(q->front == q->rear && q->tag){
        printf("Queue Full!");
        return ERROR;
    }

    q->base[q->rear++] = e;
    q->rear %= MAXQSIZE;
    q->tag = 1;
    return OK;
}

Status DeQueue(SqQueue *q,Elemtype *e){
    if(q->front == q->rear && !(q->tag)){
        printf("Queue Empty!");
        return ERROR;
    }

    *e = q->base[q->front++];
    q->front %= MAXQSIZE;
    q->tag = 0;
    return OK;
}
```
对比另一种方法(牺牲一个存储位置)
时间方面：相比于另一种方法，由于每次判断需要判断tag（front == rear的时候），以及成功增删之后需要更新tag，如果是元素较多的情形，可能会导致时间开销更大的情况，但是实际上，时间复杂度均为O(1)，时间方面的影响可以忽略不计
空间方面：tag标志方法，可以避免有一个空间无法存储元素的情况。如果是小元素，队列长度较大，对空间的节约可忽略不计；但是如果是大元素，队列较短，采取这种方法，可以避免有一个空间没有被利用，节约空间

## 3.31
```c
//双端非循环队列
typedef struct{
    char *base;
    int front,rear;
}SqQueue;

Status IsPalindrome(char *str){
    SqQueue q;

    if(!(q.base = (char*)malloc(MAXQSIZE*sizeof(char)))){
        printf("Malloc Fail!");
        return ERROR;
    }
    q.front = q.rear = 0;

    //默认不会出现字符串中无@以及str = NULL的情形
    while(*str != '@')
        Enqueue(&q,*(str++));

    q.rear--;
    while(q.front < q.rear)
        if(q.base[q.front++] != q.base[q.rear--]){
            free(q.base);
            return FALSE;
        }
            
    free(q.base);
    return TRUE;
}
```
